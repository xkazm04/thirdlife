# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/b_stale_slide.gd"
extends SceneTree

const Space = preload("res://kernel/world/terrain/world_space.gd")
const WalkCase = preload("res://kernel/world/walks/walk_case.gd")

func _initialize() -> void:
	var sp := Space.new()
	WalkCase.flat(sp, 0.0, -10, 10, -40, 0)
	WalkCase.flat(sp, -3.0, -10, 10, 0, 40)
	var feet := Vector3(0, 0, -25)
	sp.place(feet)
	sp.rest()
	var starts := 0
	var ends := 0
	var previous_mode := "ground"
	for tick in 360:
		var wish := Vector3(0, 0, 9) if tick < 30 or tick >= 140 else Vector3.ZERO
		var result := sp.drive(feet, wish, false, tick == 30)
		feet = result.pos
		starts += result.events.count("slide")
		ends += result.events.count("end_slide")
		if result.mode == "air":
			print("REPRO stale_slide: %s tick=%d transition=%s->air events=%s slide_starts=%d slide_ends=%d" % [
				"FAIL" if previous_mode == "ground" and ends > starts else "other",
				tick, previous_mode, str(result.events), starts, ends])
			break
		previous_mode = result.mode
	sp.free_all()
	quit()
