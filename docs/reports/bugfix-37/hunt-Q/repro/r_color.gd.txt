extends SceneTree

func stick_hash(rig: Node3D) -> String:
	var mi: MeshInstance3D = rig.find_child("StickMesh", true, false)
	return mi.mesh.get_meta("tl_hash")

func _initialize() -> void:
	var colors := {"stick": "car_red"}
	var changed = TLRigs.build("biped", {"mode": "skate"})
	var built = TLRigs.build("biped", {"mode": "skate", "colors": colors})
	var old := stick_hash(changed)
	changed.set_param(&"colors", colors)
	var after := stick_hash(changed)
	var reference := stick_hash(built)
	print("REPRO color: %s stick mesh: distinct build colors=%s, runtime recolor kept old mesh hash=%s, matches requested color=%s" % ["FAIL" if old != reference and after != reference else "PASS", old != reference, after == old, after == reference])
	changed.free()
	built.free()
	quit()
