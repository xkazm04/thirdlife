# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/b_pad_lift.gd"
extends SceneTree

const Landscape = preload("res://kernel/world/landscape.gd")
const DrawBase = preload("res://kernel/world/draw/draw_base.gd")

class World extends Node:
	var packs := []

class Viewer extends Node:
	var world := World.new()

func _initialize() -> void:
	call_deferred("_run")

func _run() -> void:
	TLStyle.set_headless(false)
	var view := Viewer.new()
	root.add_child(view)
	view.add_child(view.world)
	var holder := Node3D.new()
	view.world.add_child(holder)
	view.world.packs.append(holder)
	var floor_mesh := MeshInstance3D.new()
	var sheet := PlaneMesh.new()
	sheet.size = Vector2(100, 100)
	sheet.subdivide_width = 49
	sheet.subdivide_depth = 49
	floor_mesh.mesh = sheet
	floor_mesh.position.y = 5
	holder.add_child(floor_mesh)
	var body := StaticBody3D.new()
	floor_mesh.add_child(body)
	var collision := CollisionShape3D.new()
	collision.shape = sheet.create_trimesh_shape()
	body.add_child(collision)
	var rim := DrawBase.edge_height(holder, 40)
	# Same placement as view.gd:184-186 for an authored plot at y=0.
	var plot := Transform3D(Basis.IDENTITY, Vector3(0, -rim, 0))
	for explicit_height: bool in [true, false]:
		var land := Node3D.new()
		view.add_child(land)
		var cfg := {"relief": 0, "edges_now": true, "fence": false, "trees": 0,
			"broadleaf": 0, "rocks": 0, "bushes": 0, "mesas": 0, "belt": 0,
			"soft_patches": 0, "plain_clumps": 0}
		if explicit_height:
			cfg.pad_h = [0]
		Landscape.build(land, [plot], [40], [true], cfg)
		var terrain: RefCounted = land.get_meta(&"tl_terrain")
		var sp: RefCounted = land.get_meta(&"tl_space")
		var pad: float = terrain.gy(45, 0)
		var support: float = sp.ground_at(39, 0, 20)
		var drawn := plot.origin.y + rim
		print("REPRO pad_lift: %s explicit_pad=%s pack_rim_y=%.2f pad_y=%.2f collider_y_at_r39=%.2f" % [
			"FAIL" if absf(pad - drawn) > 0.01 else "control", str(explicit_height), drawn, pad, support])
		land.free()
	view.free()
	quit()
