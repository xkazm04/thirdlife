# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/b_plain_edge.gd"
extends SceneTree

const Landscape = preload("res://kernel/world/landscape.gd")

class Viewer extends Node:
	var world: Node

func _initialize() -> void:
	call_deferred("_run")

func _run() -> void:
	TLStyle.set_headless(false)
	var view := Viewer.new()
	root.add_child(view)
	var land := Node3D.new()
	view.add_child(land)
	Landscape.build(land, [Transform3D.IDENTITY], [12], [false], {
		"relief": 0, "trees": 0, "broadleaf": 0, "rocks": 0, "bushes": 0, "belt": 0,
		"mesas": 0, "plain_clumps": 0, "soft_patches": 0, "edges_now": true})
	var sp: RefCounted = land.get_meta(&"tl_space")
	var edge: float = land.get_meta(&"tl_terrain").x1()
	var feet := Vector3(edge - 2, sp.ground_at(edge - 2, 0, 10), 0)
	sp.place(feet)
	sp.rest()
	for tick in 120:
		feet = sp.drive(feet, Vector3(4.5, 0, 0), false, false).pos
	var drawn := false
	for mi in land.get_children():
		if mi is MeshInstance3D and mi.name == "WorldFarGround":
			var faces: PackedVector3Array = mi.mesh.get_faces()
			for i in range(0, faces.size(), 3):
				var hit = Geometry3D.ray_intersects_triangle(Vector3(feet.x, 10, feet.z), Vector3.DOWN, faces[i], faces[i+1], faces[i+2])
				if hit != null and absf(hit.y) < 0.01:
					drawn = true
					break
	print("REPRO plain_edge: %s edge_x=%.1f feet_x=%.1f drawn_floor_at_y0=%s support_y=%s feet_y=%.2f" % [
		"FAIL" if drawn and feet.y < -5 else "other", edge, feet.x, str(drawn), str(sp.ground_at(feet.x, feet.z, 10)), feet.y])
	view.free()
	quit()
