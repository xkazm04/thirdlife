extends SceneTree

const Terrain = preload("res://kernel/world/terrain/village_terrain.gd")

func _initialize() -> void:
	var narrow := {"pts": PackedVector2Array([Vector2(-30, 0), Vector2(-30, 2)]),
		"ys": PackedFloat32Array([2, 2]), "half": 2.0}
	var wide := {"pts": PackedVector2Array([Vector2(30, 0), Vector2(30, 2)]),
		"ys": PackedFloat32Array([20, 20]), "half": 9.0}
	var heights := PackedFloat32Array()
	for roads: Array in [[wide, narrow], [narrow, wide]]:
		var terrain := Terrain.new(7, Rect2(-50, -20, 100, 40), 1)
		terrain.relief = 0
		terrain.roads = roads
		terrain.build_grid()
		# Inside the wide road, eight metres from its centre line and 68 m from the narrow one.
		heights.append(terrain.gy(38, 0))
	print("REPRO road_width: %s expected_y=19.98 wide_first=%.2f narrow_first=%.2f" % [
		"FAIL" if absf(heights[0] - heights[1]) > 0.01 else "pass", heights[0], heights[1]])
	quit()
