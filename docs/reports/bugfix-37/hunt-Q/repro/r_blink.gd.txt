extends SceneTree

func _initialize() -> void:
	var built = TLRigs.build("biped", {"expression": "blink"})
	var runtime = TLRigs.build("biped")
	runtime.set_param(&"expression", &"blink")
	for f in range(1, 241):
		built.tick(f)
		runtime.tick(f)
	var sk: Skeleton3D = built.get_node("Model/Skeleton3D")
	var other: Skeleton3D = runtime.get_node("Model/Skeleton3D")
	var i := sk.find_bone("LeftEye")
	var height := sk.get_bone_pose_scale(i).y
	var control := other.get_bone_pose_scale(i).y
	print("REPRO blink: %s expected build-time blink eye height %.2f, got %.2f at tick 240" % ["FAIL" if absf(height - control) > 0.01 else "PASS", control, height])
	built.free()
	runtime.free()
	quit()
