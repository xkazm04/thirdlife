extends SceneTree

func _initialize() -> void:
	var late = TLRigs.build("biped")
	var control = TLRigs.build("biped")
	for f in range(1, 121):
		control.tick(f)
		if f >= 100:
			late.tick(f)
	var sk: Skeleton3D = late.get_node("Model/Skeleton3D")
	var ref_sk: Skeleton3D = control.get_node("Model/Skeleton3D")
	var eye := sk.find_bone("LeftEye")
	var actual := sk.get_bone_pose_scale(eye).y
	var expected := ref_sk.get_bone_pose_scale(eye).y
	print("REPRO spawn_face: %s frame=120 eye height for a biped spawned at 100: %.2f, expected %.2f" % ["FAIL" if absf(actual - expected) > 0.01 else "PASS", actual, expected])
	late.free()
	control.free()
	quit()
