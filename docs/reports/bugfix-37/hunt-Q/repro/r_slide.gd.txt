# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/r_slide.gd"
extends SceneTree

func _initialize() -> void:
	run.call_deferred()

func run() -> void:
	var rig = TLRigs.build("biped")
	root.add_child(rig)
	rig.play(&"slide")
	for f in range(1, 61):
		rig.tick(f)
	var sk: Skeleton3D = rig.get_node("Model/Skeleton3D")
	# Read the real generator's rigid weights; headless rendering need not retain mesh arrays.
	var geometry := TLRigBipedMesh.build_mesh(sk, TLRigBipedMesh.LOOK, true, false)
	var hand := sk.find_bone("LeftHand")
	var skin := sk.get_bone_global_pose(hand) * sk.get_bone_global_rest(hand).affine_inverse()
	var lowest := INF
	var count := 0
	for surface in geometry.surfs.values():
		for i in surface.v.size():
			if surface.b[4 * i] == hand:
				count += 1
				lowest = minf(lowest, (skin * Vector3(surface.v[i])).y)
	print("REPRO slide: %s held slide at tick 60, floor y=0, lowest of %d left-hand vertices y=%.3f m" % ["FAIL" if count > 0 and lowest > 0.05 else "PASS", count, lowest])
	rig.free()
	quit()
