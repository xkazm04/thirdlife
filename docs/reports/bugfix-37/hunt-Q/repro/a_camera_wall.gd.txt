# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/a_camera_wall.gd"
extends SceneTree

const Chase = preload("res://kernel/api/tl_chase_cam.gd")
const Space = preload("res://kernel/world/terrain/world_space.gd")

func _initialize() -> void:
	call_deferred("_check")

func _check() -> void:
	var sp = Space.new()
	# Solid wall covers z=0.2..10.2. Both look points are outside.
	sp.add_box(Transform3D(Basis(), Vector3(0, 2, 5.2)), Vector3(20, 4, 10))
	var who := Node3D.new()
	var cam = Chase.new()
	root.add_child(who)
	root.add_child(cam)
	who.position.z = -1.0
	cam.setup(who, {"distance": 7.0, "look_height": 1.0, "height": 1.0,
			"avoid_margin": 0.3, "avoid_space": sp.space})
	var st := PhysicsServer3D.space_get_direct_state(sp.space)
	var query := PhysicsPointQueryParameters3D.new()
	query.collision_mask = 1
	query.position = cam.global_position
	var control := not st.intersect_point(query).is_empty()
	who.position.z = 0.0
	cam.snap()
	query.position = cam.global_position
	var inside := not st.intersect_point(query).is_empty()
	print("REPRO camera_wall: %s expected camera outside wall, inside at 1.2 m clearance=%s, inside at 0.2 m clearance=%s, camera z=%.2f" % ["FAIL" if inside and not control else "PASS", control, inside, cam.global_position.z])
	cam.free()
	who.free()
	sp.free_all()
	quit()
