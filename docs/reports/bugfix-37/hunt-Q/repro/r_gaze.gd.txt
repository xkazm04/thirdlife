# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/r_gaze.gd"
extends SceneTree

func _initialize() -> void:
	run.call_deferred()

func run() -> void:
	var rig = TLRigs.build("biped")
	root.add_child(rig)
	var sk: Skeleton3D = rig.get_node("Model/Skeleton3D")
	var neck := sk.find_bone("Neck")
	rig.set_param(&"look_at", Vector3(100, 1.75, 0))
	rig.play(&"fall")
	var peak := 0.0
	var peak_frame := 0
	for f in range(1, 30):
		rig.tick(f)
		var angle := rad_to_deg(sk.get_bone_pose_rotation(neck).angle_to(Quaternion.IDENTITY))
		if angle > peak:
			peak = angle
			peak_frame = f
	print("REPRO gaze: %s at fall tick %d a horizontal target has neck rotation %.1f degrees, authored neck neutral, look limit 28 degrees" % ["FAIL" if peak > 28.1 else "PASS", peak_frame, peak])
	rig.free()
	quit()
