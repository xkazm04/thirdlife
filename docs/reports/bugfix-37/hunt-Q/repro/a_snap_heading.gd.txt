# timeout 120 "C:/Users/kazda/kiro/gtai-spike/tools/godot/Godot_v4.7.2-stable_win64_console.exe" --headless --path engine --script "$PWD/docs/reports/astra-trial/d5/repro/a_snap_heading.gd"
extends SceneTree

const Chase = preload("res://kernel/api/tl_chase_cam.gd")

func _initialize() -> void:
	call_deferred("_check")

func _check() -> void:
	var who := Node3D.new()
	var cam = Chase.new()
	root.add_child(who)
	root.add_child(cam)
	cam.setup(who, {"distance": 8.0, "stiffness": 1.0})
	who.position = Vector3(15, 0, 10)
	who.rotation.y = PI / 2
	cam.snap()
	var behind := who.global_basis.z
	var offset: Vector3 = cam.global_position - who.global_position
	var back_m := offset.dot(behind)
	print("REPRO snap_heading: %s expected 8.0 m behind the teleported target immediately, got %.1f m" % ["PASS" if is_equal_approx(back_m, 8.0) else "FAIL", absf(back_m)])
	quit()
