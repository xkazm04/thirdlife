# bash docs/shifts/2026-11-19-night/lanes/hunt2/repro/run.sh B12
extends SceneTree

func _initialize() -> void:
	var missing: String = get_script().resource_path.get_base_dir() + "/absent-style.json"
	var errors := TLStyle.load_style(missing)
	if errors.is_empty():
		quit(2)
		return
	var fallback := TLStyle.palette().size()
	var defaults := TLStyle.load_style(TLStyle.DEFAULT_PATH)
	var expected := TLStyle.palette().size()
	if expected < 1 or not defaults.is_empty():
		quit(2)
		return
	var again := TLStyle.load_style(missing)
	var warm := TLStyle.palette().size()
	if again.is_empty() or warm != expected:
		quit(2)
		return
	print("REPRO B12: FAIL first rejected style leaves %d palette keys; default %d, rejected reload after default %d" % [fallback, expected, warm] if fallback != expected else "REPRO B12: PASS")
	quit(int(fallback != expected))
