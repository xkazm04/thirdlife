# bash docs/shifts/2026-11-19-night/lanes/hunt2/repro/run.sh A3
extends SceneTree

const Pack = preload("res://kernel/world/pack.gd")

func _initialize() -> void:
	run.call_deferred()

func run() -> void:
	var pack := Pack.new()
	root.add_child(pack)
	if not pack.load_pack("res://packs/parkour", {"own_space": false, "scenery": "plot"}):
		printerr(pack.load_errors)
		quit(2)
		return
	var race = pack.entry
	var runner: TLBiped = race.runner
	var venue = race.venue
	# Earn each preceding gate through the real pack tick, then stand clear of the beam.
	pack.tick(0)
	var tick := 1
	for station: float in [venue.checkpoints[0] + 0.1, venue.checkpoints[1] + 0.1, venue.bar - 0.55]:
		runner.place(venue.point(station, venue.DECK_Y + 0.01))
		for i in 5:
			await physics_frame
			pack.tick(tick)
			tick += 1
	var before := runner.position.x
	var gap: float = venue.point(venue.bar, 0).x - before - runner.body_radius()
	if gap < 0.1 or not runner.grounded or runner.speed() > 0.01 or race.rules.next_checkpoint.runner != 2:
		printerr("fixture did not settle clear of the beam")
		quit(2)
		return
	pack.ctx.input.tap(&"slide")
	pack.tick(tick)
	var moved := before - runner.position.x
	var text := "REPRO A3: PASS"
	if moved > 1:
		text = "REPRO A3: FAIL C sent a stationary runner %.2f m backward with %.2f m of clear air between capsule and beam" % [moved, gap]
	pack.free()
	print(text)
	quit(int(moved > 1))
