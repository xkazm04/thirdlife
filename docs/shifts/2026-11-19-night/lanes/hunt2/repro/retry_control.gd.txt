# bash docs/shifts/2026-11-19-night/lanes/hunt2/repro/run.sh retry_control
extends SceneTree

const Pack = preload("res://kernel/world/pack.gd")

func _initialize() -> void:
	run.call_deferred()

func run() -> void:
	var pack := Pack.new()
	root.add_child(pack)
	if not pack.load_pack("res://packs/parkour", {"own_space": false, "scenery": "plot"}):
		printerr(pack.load_errors)
		quit(2)
		return
	var race = pack.entry
	var runner: TLBiped = race.runner
	var venue = race.venue
	pack.tick(0)
	var tick := 1
	var guards: Array[bool] = []
	# Stand on the finish stripe before gates 1 and 2, then earn those gates in order.
	for index: int in [3, 0, 1]:
		var h: float = 0 if index == 3 else venue.DECK_Y
		runner.place(venue.point(venue.checkpoints[index] + 0.1, h + 0.01))
		for i in 3:
			await physics_frame
			pack.tick(tick)
			tick += 1
		guards.append(race.rules.next_checkpoint.runner == (0 if index == 3 else index + 1))
	# Let the actual body step begin the fall before C is sampled.
	runner.place(venue.point(venue.bar - 0.4, venue.DECK_Y + 0.9))
	for i in 3:
		await physics_frame
		pack.tick(tick)
		tick += 1
	if runner.grounded or runner.velocity.y >= 0:
		printerr("failed to stage a falling walker")
		quit(2)
		return
	var before := runner.position.x
	pack.ctx.input.tap(&"slide")
	pack.tick(tick)
	tick += 1
	guards.append(is_equal_approx(runner.position.x, before) and race.rules.next_checkpoint.runner == 2)
	# First show that a blocked runner is sent back; fresh presses there must not advance a gate.
	runner.place(race.spawn_at)
	pack.ctx.input.press(&"forward")
	pack.ctx.input.press(&"run")
	for i in 65:
		await physics_frame
		pack.tick(tick)
		tick += 1
	# A15 separately tests the ticks on which the retry is refused. This control
	# needs one accepted retry first, to check what five fresh taps do afterwards.
	for i in 12:
		await physics_frame
		if runner.speed() < 0.5:
			break
		pack.tick(tick)
		tick += 1
	var bar_x := runner.position.x
	pack.ctx.input.release(&"forward")
	pack.ctx.input.release(&"run")
	pack.ctx.input.tap(&"slide")
	pack.tick(tick)
	tick += 1
	var landing_x := runner.position.x
	guards.append(bar_x - landing_x > 3)
	for i in 10:
		await physics_frame
		if i % 2 == 1:
			pack.ctx.input.tap(&"slide")
		pack.tick(tick)
		tick += 1
	guards.append(race.rules.next_checkpoint.runner == 2 and absf(runner.position.x - landing_x) < 0.1)
	# Complete the clock at the trigger location, preserving gate 2; C must have no effect.
	runner.place(venue.point(venue.bar - runner.body_radius(), venue.DECK_Y + 0.01))
	for i in 4:
		await physics_frame
		pack.tick(tick)
		tick += 1
	race.rules.finish("", "time")
	before = runner.position.x
	pack.ctx.input.tap(&"slide")
	pack.tick(tick)
	guards.append(is_equal_approx(before, runner.position.x) and race.hud.text_of(&"race").begins_with("TIME UP"))
	var bad := guards.count(false)
	pack.free()
	print("guards (skip, gate 1, gate 2, fall, contact retry, 5 more taps, timer over) = ", guards)
	print("REPRO retry_control: PASS" if bad == 0 else "REPRO retry_control: FAIL %d/7 guards failed" % bad)
	quit(int(bad > 0))
