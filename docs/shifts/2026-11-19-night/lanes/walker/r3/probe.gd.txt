extends SceneTree
## Windowed measurement. Run through capture.py. The host/review scene still owns physics and camera.
var output := OS.get_environment("INK_OUTPUT")
var place := OS.get_environment("INK_PLACE")

func _initialize() -> void:
	call_deferred("measure")

func dump(label: String) -> void:
	for i in 3:
		if place in ["bluff", "bank"]:
			RenderingServer.force_draw(false, 0.0)
		else:
			await RenderingServer.frame_post_draw
	var img := root.get_texture().get_image()
	if img.save_png(output + "/" + place + "-" + label + ".png") != OK:
		push_error("capture failed")
		quit(2)

func measure() -> void:
	var village := place in ["bluff", "bank"]
	var scene: Node
	if village:
		await process_frame
		scene = current_scene
		if scene == null:
			push_error("world host main scene missing")
			quit(1)
			return
	else:
		scene = load("res://rigs/review/review.tscn").instantiate()
		root.add_child(scene)
	var driver: Node = scene.get_node("WorldView/World") if village else scene
	var at := 310 if place == "bank" else 120 if village else 100 if place == "frozen" else 75
	var counter := "tick" if village else "frame"
	while int(driver.get(counter)) < at:
		await process_frame
	paused = not village
	var cam := root.get_camera_3d()
	var rig: Node3D = scene.get_node("WorldView").visitor.rig if village else scene.rig
	print("sample physics=%s camera=%s rig=%s" % [driver.get(counter), cam.global_position, rig.global_position])
	if place.ends_with("_wall"):
		# Vertical review wall, 0.65 m beyond the rig origin along the horizontal sight line.
		var axis := -cam.global_basis.z
		axis.y = 0
		axis = axis.normalized()
		var wall := MeshInstance3D.new()
		var box := BoxMesh.new()
		box.size = Vector3(12, 6, 0.1)
		wall.mesh = box
		wall.material_override = TLStyle.flat_material(TLStyle.color("concrete"))
		scene.add_child(wall)
		wall.global_position = rig.global_position + axis * 0.70 + Vector3.UP * 2
		wall.look_at(wall.position + axis)
		wall.reset_physics_interpolation()
	if place == "grazing":
		cam.look_at_from_position(Vector3(25, 1.4, 28), Vector3(60, 0, 28))
		cam.reset_physics_interpolation()
	var post: MeshInstance3D = scene.find_child("InkPost", true, false)
	var mat: ShaderMaterial = post.material_override
	if OS.has_environment("INK_BASELINE"):
		var previous := Shader.new()
		previous.code = FileAccess.get_file_as_string(OS.get_environment("INK_BASELINE"))
		mat.shader = previous
		TLStyle.push("ink", mat)
	if not OS.has_environment("INK_PLANE"):
		await dump("beauty")
	var we: WorldEnvironment = post.get_parent().get_node("WorldEnvironment")
	var env := we.environment
	env.tonemap_mode = Environment.TONE_MAPPER_LINEAR
	env.tonemap_exposure = 1
	env.glow_enabled = false
	we.camera_attributes = null
	root.screen_space_aa = Viewport.SCREEN_SPACE_AA_DISABLED
	for layer in scene.find_children("*", "CanvasLayer", true, false):
		layer.visible = false
	mat.render_priority = 127
	# Keep the production shader and uniforms. Replace only its final output, in memory.
	var original := mat.shader.code.strip_edges().trim_suffix("}")
	if OS.has_environment("INK_PLANE"):
		var begin := original.find("float background_gap =")
		var end := original.find("ne = max(ne,", begin)
		if begin < 0 or end < 0:
			push_error("missing inner-ring measurement site")
			quit(2)
			return
		var variant_code := """
		vec3 ray = ink_ray(juv, VIEWPORT_SIZE, INV_PROJECTION_MATRIX);
		core = max(core, plane_edge(n0, ray, ink_ray(juv + dir * px * wi, VIEWPORT_SIZE, INV_PROJECTION_MATRIX), d0, dbp));
		core = max(core, plane_edge(n0, ray, ink_ray(juv - dir * px * wi, VIEWPORT_SIZE, INV_PROJECTION_MATRIX), d0, dbm));
		"""
		original = original.substr(0, begin) + variant_code + original.substr(end)
		original = original.replace("void fragment()", FileAccess.get_file_as_string(OS.get_environment("INK_PLANE")) + "\nvoid fragment()")
	var shader := Shader.new()
	shader.code = original.replace("void fragment()", "uniform bool measure_depth = false;\nvoid fragment()") + """
	ALPHA = 1.0;
	ALBEDO = vec3(step(0.5,body),step(0.5,outer*0.96),step(graze,0.2));
	if (measure_depth) {
		// Centimetres packed in base 16. Each sRGB channel has only 16 levels.
		float cm = clamp(round(lin_depth(uv, INV_PROJECTION_MATRIX)*100.0),0.0,4095.0);
		vec3 digits = mod(floor(cm / vec3(256.,16.,1.)),16.)/15.;
		ALBEDO = mix(digits/12.92, pow((digits + 0.055)/1.055,vec3(2.4)),step(vec3(0.04045),digits));
	}
}
"""
	mat.shader = shader
	TLStyle.push("ink", mat)
	await dump("signal")
	mat.set_shader_parameter("measure_depth", true)
	await dump("front")
	rig.hide()
	await dump("back")
	mat.set_shader_parameter("measure_depth", false)
	await dump("hidden")
	print("INK_SAMPLE_DONE ", place)
	if village:
		paused = false # Let the world finish its scheduled ticks and release pack resources.
	else:
		quit()
