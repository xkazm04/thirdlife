// keys.mjs <game> <frames> <seed> : a newcomer's key stream for TL_EXE_KEYS (frame:+Key / frame:-Key; frame:~dx,dy moves the
// mouse), open loop, after J2nd's keys.mjs. He presses only keys the screen has shown him (read from the pack's hint code
// and checked on film), late (a reaction delay after the screen says it), in uneven bursts, and misses some.
const [game, N, seed] = [process.argv[2], +process.argv[3], +(process.argv[4] || 1)];
let s = seed * 9301 + 49297; const r = () => (s = (s * 9301 + 49297) % 233280) / 233280; const ri = (a, b) => a + Math.floor(r() * (b - a + 1));
const ev = []; const put = (f, k, down) => { if (f < N) ev.push([f, (down ? '+' : '-') + k]); };
const tap = (f, k, len = 6) => { put(f, k, true); put(f + len, k, false); };
const bursts = (keys, gapA, gapB, lenA, lenB, from = 150, to = N) => { for (let f = from + ri(0, gapB); f < to; f += ri(gapA, gapB)) { const k = keys[ri(0, keys.length - 1)]; const l = ri(lenA, lenB); put(f, k, true); put(f + l, k, false); f += l; } };
const taps = (k, gapA, gapB, from = 150, len = 6, to = N, miss = 0.2) => { for (let f = from + ri(gapA / 2, gapB); f < to; f += ri(gapA, gapB)) if (r() > miss) tap(f, k, len); };
const holds = (k, gapA, gapB, lenA, lenB, from, to = N, miss = 0.2) => { for (let f = from + ri(0, gapB); f < to; f += ri(gapA, gapB)) { if (r() < miss) continue; const l = ri(lenA, lenB); put(f, k, true); put(f + l, k, false); f += l; } };
const F1 = +(process.argv[5] || 0);   // the frame a later cue appears (from film), 0 = not yet seen
if (game === 'robot') {
  // told: "Space: slide" (the opening), "Hold Space: shoot" with the ball, "Hold A or D: pick a corner"; WASD is the
  // universal guess; "Hold F, let go: SMASH" only after the opening (F1 = when the film shows it)
  put(150, 'W', true);
  bursts(['A', 'D', 'A', 'D', 'S'], 40, 150, 15, 70);
  taps('Space', 90, 220, 200);
  holds('Space', 300, 600, 30, 70, 400);
  if (F1) holds('F', 200, 420, 40, 100, F1 + ri(40, 90));
} else if (game === 'shield') {
  // told from frame 0: "Hold RIGHT MOUSE: shield up" (he holds Space, the pack's key for it: TL_EXE_KEYS has no mouse
  // buttons), then "Now bash: keep holding, click LEFT MOUSE" (J, the pack's key); the key row shows Q CALL, F, C, G READY
  bursts(['W', 'W', 'A', 'D', 'W', 'S'], 30, 120, 20, 90, 120);
  holds('Space', 90, 200, 60, 180, 180);
  taps('J', 40, 110, 260);
  if (F1) { taps('F', 300, 600, F1, 6, N, 0.3); taps('C', 400, 700, F1 + 120, 6, N, 0.3); taps('G', 500, 900, F1 + 200, 6, N, 0.3); taps('Q', 500, 900, F1 + 60, 6, N, 0.3); }
} else if (game === 'hockey') {
  // told: PRESS ENTER (the front), "WASD skates, the mouse is your stick"; in the match "F: METEOR Light your stick, then
  // hold shoot to wind up". The exe cannot press mouse buttons, so he never shoots here (said in PROOF).
  tap(120, 'Enter', 8);
  if (F1) tap(F1, 'Enter', 8);
  bursts(['W', 'W', 'A', 'D', 'W', 'S', 'D'], 20, 100, 25, 120, 300);
  for (let f = 320; f < N; f += ri(8, 30)) ev.push([f, `~${ri(-40, 40)},${ri(-30, 30)}`]);
  taps('F', 300, 600, 600, 6, N, 0.3);
  holds('Shift', 300, 600, 40, 120, 700);
} else if (game === 'kart') {
  // told: "W drive, SHIFT boost, SPACE bump its side" from the start; after a box "F fires a red shell"; then "Q jumps
  // over one" (J2nd round 2, v0.1.11's lesson hint line); E (whirl) is never named in the first minute
  put(140, 'W', true);
  bursts(['A', 'D'], 50, 170, 15, 60);
  taps('Space', 90, 240); taps('Shift', 300, 500, 300, 40);
  taps('F', 240, 420, 700); taps('Q', 400, 700, 1200);
} else if (game === 'plane') {
  // told: "HOLD W", "SPACE: fire when they sit in your sight", the lesson's "Q: HARPOON her as she crosses your nose";
  // the roll (Shift) is named by her "HARPOON! ROLL" warning; E (loop) and F are not named in the first minute
  put(140, 'W', true);
  bursts(['Left', 'Right', 'Up', 'Down', 'Left', 'Right'], 30, 110, 20, 80);
  taps('Space', 40, 100, 150, 10); taps('Q', 300, 450, 300); taps('Shift', 400, 700, 900);
} else if (game === 'court') {
  // told: "HOLD E ON THE CHALK", "SPACE: SWAT AS THEY RISE", SPACE to shoot; WASD the universal guess; C is never named
  bursts(['W', 'A', 'S', 'D', 'W', 'W'], 20, 90, 30, 120);
  for (let f = 300 + ri(0, 150); f < N; f += ri(150, 300)) { put(f, 'Space', true); put(f + ri(18, 40), 'Space', false); }
  holds('E', 300, 600, 40, 120, 400);
}
ev.sort((a, b) => a[0] - b[0]);
console.log(ev.map(([f, k]) => `${f}:${k}`).join(';'));
