extends Node
## Lane S4-core (27 Nov, spike): a whole match (the cartridge's 3-minute timer, the bots at data/match.json's level) played
## by bot_meteor's scripted decisions through a person's hand (TLHand: its reaction, tap lengths, late releases, lapses and
## cursor). Its decisions: chase the loose puck with the cursor on it; carrying, skate at the net, F when the meteor is ready
## within 30 m (scripts/meteor.gd says so), a full wind-up from 26 m while lit, an 18-frame shot inside 14 m; an opponent
## carrying within 2.5 m: the body check or the poke; at a faceoff the left button at the drop. A bot_ladder_<level>_<hand>.gd
## names the profile. Its result is `tl ladder`'s: won, goals, against, frames, and the bots' side of the meteor (s4_bots.gd).
## A hand measures the game through that hand: quote a figure with its profile.

const REACT: int = 12
const METEOR_M: float = 26.0
const WIND_UP: int = 46
const MAX_FRAMES: int = 16000         # the 10,800-frame timer plus goal pauses and faceoffs
const SKATE: Array = [&"k_w", &"k_s", &"k_a", &"k_d"]

var ctx
var hand: TLHand
var profile: String = "regular"
var level: String = "rookie"
var decided: int = -100000
var keys: String = ""
var shot_hold: int = 0
var f_wait: int = 0
var check_wait: int = 0
var shots: int = 0
var checks: int = 0
var f_presses: int = 0
var carried_frames: int = 0
var done: bool = false
var net_x: float = 26.0


func ladder(lv: String, prof: String) -> void:
	level = lv
	profile = prof


func _bot_setup(c) -> void:
	ctx = c
	hand = TLHand.create(ctx, profile)
	var m: Variant = ctx.read_json("match.json")
	var cart: Variant = ctx.read_json(String((m as Dictionary).get("cartridge", "hockey.regular.cartridge.json"))) if m is Dictionary else null
	if cart is Dictionary and (cart as Dictionary).get("game") is Dictionary and (cart["game"] as Dictionary).get("rink") is Dictionary:
		net_x = maxf(4.0, float((cart["game"]["rink"] as Dictionary).get("length_m", 90.0)) * 0.5 - 4.0)


func _keys_toward(s: Dictionary, to: Vector2) -> String:
	if to.length() < 0.5:
		return ""
	var want: Vector2 = to.normalized()
	var cy: float = deg_to_rad(float(s["cam_yaw"]))
	var a: float = atan2(want.dot(Vector2(-sin(cy), cos(cy))), want.dot(Vector2(cos(cy), sin(cy))))
	var a8: float = roundf(a / (PI / 4.0)) * (PI / 4.0)
	var kx: float = sin(a8)
	var ky: float = cos(a8)
	return ("W" if ky > 0.3 else "") + ("S" if ky < -0.3 else "") + ("D" if kx > 0.3 else "") + ("A" if kx < -0.3 else "")


func _bot_tick(frame: int) -> void:
	var s: Dictionary = ctx.pack_state()
	if frame >= MAX_FRAMES or bool(s.get("over", false)):
		done = true
		hand.release_all()
		hand.tick()
		return
	var me := Vector2(float(s["x"]), float(s["y"]))
	var puck := Vector2(float(s["puck_x"]), float(s["puck_y"]))
	var carrier: int = int(s["carrier"])
	var net := Vector2(net_x if int(s["team"]) == 0 else -net_x, 0.0)
	var target: Vector2 = puck
	var aim: Vector2 = puck
	var sprint: bool = false
	var met: Dictionary = ctx.root.get_meta(&"meteor", {})
	if f_wait > 0:
		f_wait -= 1
	if check_wait > 0:
		check_wait -= 1
	if int(s["phase"]) == 3:
		target = me
		if int(s["faceoff"]) <= 1 and shot_hold == 0:
			hand.tap(&"m_left", 2)
	elif carrier == int(s["human"]):
		carried_frames += 1
		target = net
		aim = net
		sprint = me.distance_to(net) > 20.0
		if bool(met.get("armed_now", false)) and me.distance_to(net) < METEOR_M and shot_hold == 0:
			shot_hold = WIND_UP
		elif me.distance_to(net) < 14.0 and shot_hold == 0:
			shot_hold = 18
		if f_wait == 0 and bool(met.get("ready_now", false)) and int(s["phase"]) == 0 and me.distance_to(net) < METEOR_M + 4.0:
			hand.tap(&"k_f", 2)
			f_presses += 1
			f_wait = 30
	elif carrier >= 0 and me.distance_to(puck) < 2.5 and check_wait == 0:
		hand.tap(&"m_right" if frame % 2 == 0 else &"k_space", 2)
		checks += 1
		check_wait = 20
	else:
		sprint = me.distance_to(puck) > 15.0
	if frame - decided >= REACT:
		keys = _keys_toward(s, target - me)
		decided = frame
	for i: int in 4:
		if keys.contains(["W", "S", "A", "D"][i]):
			hand.press(SKATE[i])
		else:
			hand.release(SKATE[i])
	if sprint:
		hand.press(&"k_shift")
	else:
		hand.release(&"k_shift")
	hand.set_aim(&"aim", Vector3(aim.x, 0.0, aim.y))
	if shot_hold > 0:
		hand.press(&"m_left")
		shot_hold -= 1
		if shot_hold == 0:
			shots += 1
			hand.release(&"m_left")
	hand.tick()


func _bot_done() -> bool:
	return done


func _bot_result() -> Dictionary:
	var s: Dictionary = ctx.pack_state()
	var score: Array = s["rink"]["score"]
	var mine: int = int(s.get("team", 0))
	var gf: int = int(score[mine]) if score.size() > mine else 0
	var ga: int = int(score[1 - mine]) if score.size() > 1 - mine else 0
	var s4: Dictionary = ctx.root.get_meta(&"s4", {})
	var met: Dictionary = ctx.root.get_meta(&"meteor", {})
	return {"ok": gf > ga, "won": gf > ga, "goals": gf, "against": ga, "frames": ctx.frame, "shots": shots,
		"summary": "%s hand at %s: %d:%d in %d frames, %d shots, %d F, s4 %s, meteor uses %d, core %s" % [profile, level, gf, ga, ctx.frame, shots, f_presses,
			JSON.stringify(s4), int(met.get("uses", 0)), String(s["rink"]["hash"])],
		"metrics": {"s4": s4, "meteor": met, "hand": hand.stats(), "carried_frames": carried_frames, "checks": checks, "f_presses": f_presses, "core_hash": String(s["rink"]["hash"])}}
